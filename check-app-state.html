Open dev console (Cmd+Option+I) and paste:

// Check all localStorage and sessionStorage
console.log('=== STORAGE STATE ===');
console.log('localStorage:');
for (let i = 0; i < localStorage.length; i++) {
    const key = localStorage.key(i);
    console.log(key + ':', localStorage.getItem(key));
}
console.log('\nsessionStorage:');
for (let i = 0; i < sessionStorage.length; i++) {
    const key = sessionStorage.key(i);
    console.log(key + ':', sessionStorage.getItem(key));
}

// Check setup wizard state
console.log('\n=== SETUP WIZARD STATE ===');
try {
    const setupData = JSON.parse(localStorage.getItem('setup-wizard-storage') || '{}');
    console.log('Setup wizard data:', setupData);
    console.log('isComplete:', setupData.state?.isComplete);
} catch (e) {
    console.log('Error parsing setup wizard:', e);
}

// Try to clear everything and reload
console.log('\n=== TO FIX: Run these commands ===');
